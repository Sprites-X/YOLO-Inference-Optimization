#   python3 -m venv .venv && source .venv/bin/activate
#   pip install --upgrade pip setuptools wheel packaging
#
#   # 1. PyTorch — ต้องระบุ index ของ pytorch.org ไม่งั้นจะได้ตัว CPU
#   #    cu128 ไม่ใช่ cu130: onnxruntime-gpu release ยังเป็น CUDA 12.x
#   #    ถ้าใช้ torch cu130 จะได้ nvidia lib สองเวอร์ชันปนกันใน venv เดียว
#   pip install torch==2.11.0+cu128 torchvision==0.26.0+cu128 \
#       --index-url https://download.pytorch.org/whl/cu128
#
#   # 2. TensorRT — ต้องใช้ index ของ NVIDIA
#   #    PyPI เสิร์ฟ sdist ที่ค้างเงียบตอน build metadata (ดึง lib 4.3GB
#   #    โดยไม่มี progress bar) index ของ NVIDIA เสิร์ฟ wheel จริง
#   pip install --extra-index-url https://pypi.nvidia.com \
#       tensorrt-cu12==10.16.1.11 \
#       tensorrt-cu12-libs==10.16.1.11 \
#       tensorrt-cu12-bindings==10.16.1.11
#
#   # 3. ที่เหลือ
#   pip install -r requirements.txt
#
#   python verify_env.py     # ต้องไม่มี FAIL ก่อนเริ่มวัด

# --- inference runtimes --------------------------------------------------
# ต้องเป็น build ที่มี CUDA EP จริง — verify_env.py เช็กว่าไม่ตกไป CPU เงียบๆ

# (execute_async_v3 + IInt8EntropyCalibrator2 ซึ่งอาจถูกถอดใน 11)
# ต้องลงจาก --extra-index-url https://pypi.nvidia.com ตามด้านบน

# TRT 10 sample ใช้ cuda-python (cudart) ไม่ใช่ pycuda แล้ว
