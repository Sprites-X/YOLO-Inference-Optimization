# requirements.txt — intent: what is needed at minimum, and which index it comes from.
#
# To reproduce the exact environment, use requirements.lock.txt instead.
# This file says why each package is here; the lock file says which versions were used.
#
# ─────────────────────────────────────────────────────────────────────
# Install order (matters — the first two steps do not come from ordinary PyPI)
#
#   python3 -m venv .venv && source .venv/bin/activate
#   pip install --upgrade pip setuptools wheel packaging
#
#   # 1. PyTorch — the pytorch.org index must be given, or you get the CPU build.
#   #    cu128 not cu130: the onnxruntime-gpu release is still CUDA 12.x, and
#   #    torch cu130 would put two nvidia lib versions in the same venv.
#   pip install torch==2.11.0+cu128 torchvision==0.26.0+cu128 \
#       --index-url https://download.pytorch.org/whl/cu128
#
#   # 2. TensorRT — needs NVIDIA's index.
#   #    PyPI serves an sdist that hangs silently while building metadata
#   #    (pulling 4.3GB of libs with no progress bar). NVIDIA's index serves
#   #    real wheels.
#   pip install --extra-index-url https://pypi.nvidia.com \
#       tensorrt-cu12==10.16.1.11 \
#       tensorrt-cu12-libs==10.16.1.11 \
#       tensorrt-cu12-bindings==10.16.1.11
#
#   # 3. Everything else
#   pip install -r requirements.txt
#
#   python verify_env.py     # must report zero failures before measuring
# ─────────────────────────────────────────────────────────────────────

# --- inference runtimes --------------------------------------------------
# Must be a build with a real CUDA EP — verify_env.py checks it does not quietly
# fall back to CPU.
onnxruntime-gpu>=1.23.0

# TensorRT 10.8+ only (Blackwell sm_120).
# Pinned because TRT 11 is out and this code uses the TRT 10 API
# (execute_async_v3 + IInt8EntropyCalibrator2, which may be gone in 11).
# Install from --extra-index-url https://pypi.nvidia.com as above.
tensorrt-cu12==10.16.1.11

# TRT 10 samples use cuda-python (cudart) rather than pycuda.
# <13 because torch cu128 pins cuda-bindings<13 — left free, pip installs 13.x and they clash.
cuda-python<13

# --- model and export ----------------------------------------------------
ultralytics>=8.4.0
onnx>=1.22.0
onnxslim>=0.1.48        # ultralytics moved from onnxsim to this

# --- image processing / evaluation ---------------------------------------
# opencv 5.x is tested against common.py (letterbox + bbox mapping agree).
# Fall back to opencv-python==4.10.0.84 if it causes trouble.
opencv-python>=4.10.0
numpy>=1.26.0
pycocotools>=2.0.11     # >=2.0.11 is the first version compatible with numpy 2.x
matplotlib>=3.8.0       # make_report.py uses it for the figures

# --- notes ---------------------------------------------------------------
# torch / torchvision are deliberately not listed here — they have to come from
# the pytorch.org index first (see step 1). List them and pip pulls the CPU build
# from PyPI over the top.
#

# GA tar from developer.nvidia.com (pick CUDA 12.x and TAR, not DEB) and add only
# $TRT_ROOT/bin to PATH — do not add LD_LIBRARY_PATH, it shadows the .so files
# pip's tensorrt import relies on.
